# Pad kernel build configuration.
#
# Scope: Ascend910B3 only. The public call chain is:
#   asnumpy.pad -> Python wrapper -> pybind -> aclnnPad/aclnnPadGetWorkspaceSize -> Pad kernel.
#
# The kernel entry is generated by the kernel stage and must expose the fixed Ascend C signature:
#   __global__ __aicore__ void pad(GM_ADDR input, GM_ADDR paddings, GM_ADDR output,
#                                  GM_ADDR workspace, GM_ADDR tiling)
# The implementation must use Init -> CopyIn -> Compute -> CopyOut and must not silently
# fall back to host computation.

if(NOT DEFINED package_name OR package_name STREQUAL "")
    message(FATAL_ERROR
        "package_name must be defined by the root CMakeLists.txt before adding op_kernel")
endif()

if(NOT DEFINED ASCEND_COMPUTE_UNIT OR ASCEND_COMPUTE_UNIT STREQUAL "")
    message(FATAL_ERROR
        "ASCEND_COMPUTE_UNIT must be set; this Pad draft supports only Ascend910B3 "
        "(ASCEND_COMPUTE_UNIT=ascend910b)")
endif()

if(NOT ASCEND_COMPUTE_UNIT STREQUAL "ascend910b")
    message(FATAL_ERROR
        "Unsupported ASCEND_COMPUTE_UNIT='${ASCEND_COMPUTE_UNIT}' for Pad. "
        "Only Ascend910B3 (ASCEND_COMPUTE_UNIT=ascend910b) is in scope for this draft.")
endif()

# The deterministic kernel stage uses the operator stem 'pad'. Accept either the
# architecture-suffixed file used by the CANNBot template or the plain stem file.
set(PAD_KERNEL_FILE "")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/pad.cpp")
    set(PAD_KERNEL_FILE "pad.cpp")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/pad_arch22.cpp")
    set(PAD_KERNEL_FILE "pad_arch22.cpp")
else()
    message(FATAL_ERROR
        "Pad kernel source is missing in ${CMAKE_CURRENT_SOURCE_DIR}; "
        "expected pad.cpp or pad_arch22.cpp")
endif()

npu_op_kernel_sources(ascendc_kernels
    OP_TYPE Pad
    KERNEL_DIR .
    KERNEL_FILE ${PAD_KERNEL_FILE}
)

npu_op_kernel_library(ascendc_kernels
    SRC_BASE ${CMAKE_CURRENT_SOURCE_DIR}
    TILING_LIBRARY cust_optiling
)

npu_op_package_add(${package_name}
    LIBRARY
        ascendc_kernels
)
